func TestKentBareStructuredWebsocketError(t *testing.T) {
	event, err := receiveKentWebsocketFrame(t, `{"error":{"code":"arbitrary-machine-code","message":"provider failure","type":"quota"}}`)
	if err != nil {
		t.Fatal(err)
	}
	failure := event.OfResponsesServerEventResponseWsError
	if failure.Error.Code != "arbitrary-machine-code" || failure.Error.Type != "quota" {
		t.Fatalf("structured error fields lost: %+v", failure.Error)
	}
	if failure.JSON.Status.Valid() || failure.Error.JSON.Param.Valid() {
		t.Fatal("absent fields became reported facts")
	}
}

func TestKentWebsocketEnvelopeValidation(t *testing.T) {
	for _, test := range []struct {
		name      string
		frame     string
		wantError bool
	}{
		{"ordinary", `{"type":"response.output_text.delta","output_index":0,"content_index":0,"delta":"text","sequence_number":1}`, false},
		{"typed error", `{"type":"error","status":402,"error":{"code":"different-code","message":"failure"}}`, false},
		{"malformed", `{"error":`, true},
		{"invalid type", `{"type":42,"error":{"code":"code","message":"failure"}}`, true},
		{"empty explicit type", `{"type":"","error":{"code":"code","message":"failure"}}`, true},
		{"null explicit type", `{"type":null,"error":{"code":"code","message":"failure"}}`, true},
		{"invalid error", `{"error":{"code":42,"message":"failure"}}`, true},
		{"missing error", `{"stream_id":"lane"}`, true},
	} {
		t.Run(test.name, func(t *testing.T) {
			event, err := receiveKentWebsocketFrame(t, test.frame)
			if (err != nil) != test.wantError {
				t.Fatalf("receive error = %v, expected error presence %t", err, test.wantError)
			}
			if err == nil && event.RawJSON() != test.frame {
				t.Fatal("ordinary typed event bytes changed")
			}
		})
	}
}

func receiveKentWebsocketFrame(t *testing.T, frame string) (responses.ResponsesServerEventUnion, error) {
	t.Helper()
	ctx, cancel := context.WithTimeout(t.Context(), 5*time.Second)
	defer cancel()
	server := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		socket, err := websocket.Accept(w, r, nil)
		if err != nil {
			t.Error(err)
			return
		}
		defer socket.CloseNow()
		if err := socket.Write(ctx, websocket.MessageText, []byte(frame)); err != nil {
			t.Error(err)
			return
		}
		_, _, _ = socket.Read(ctx)
	}))
	defer server.Close()
	service := responses.NewResponseService(option.WithBaseURL(server.URL), option.WithAPIKey("test"), option.WithMaxRetries(0))
	connection, err := service.Connect(ctx, responses.ResponseConnectionOptions{})
	if err != nil {
		t.Fatal(err)
	}
	defer connection.Abort()
	return connection.Recv(ctx)
}
